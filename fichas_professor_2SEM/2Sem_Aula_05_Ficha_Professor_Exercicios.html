✅ Gabarito

def cosine_sim(a,b):
  dot=sum(x*y for x,y in zip(a,b))
  n1=math.sqrt(sum(x*x for x in a)); n2=math.sqrt(sum(y*y for y in b))
  return dot/(n1*n2) if n1*n2>0 else 0
